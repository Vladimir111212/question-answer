import pygame
from random import randint
BLACK = (1, 2, 2) 
class TextArea():
    def __init__(self, x=0, y=0, width=10, height=10, color=None):
        self.rect = pygame.Rect(x, y, width, height)
        self.fill_color = color


    def set_text(self, text, fsize=12, text_color=BLACK):
        self.text = text
        self.image = pygame.font.Font(None, fsize).render(text, True, text_color)


    def draw(self, shift_x=0, shift_y=0):
        pygame.draw.rect(mw, self.fill_color, self.rect)
        mw.blit(self.image, (self.rect.x + shift_x, self.rect.y + shift_y))


pygame.init()
back = (142, 139, 206)
mw = pygame.display.set_mode((500, 500))
mw.fill(back)
clock = pygame.time.Clock()

square1 = TextArea(50, 50, 300, 30, (199, 137, 248))
square1.set_text('Вопрос', 20, (0, 0, 0))
square1.draw(10, 10)
square2 = TextArea(50, 100, 300, 30, (199, 137, 248))
square2.set_text('Ответ', 20, (0, 0, 0))
square2.draw(10, 10)
while True:
    for event in pygame.event.get():
        if event.type == pygame.KEYDOWN:
            if event.key == pygame.K_q:
                bal = randint(1, 3)
                if bal == 1:
                    square1.set_text('Какой ваш любимый язык?', 35)
                elif bal == 2:
                    square1.set_text('Какой фрукт вы любите?', 35)
                elif bal == 3:
                    square1.set_text('ваша нелюбимая игра?', 35)
                square1.draw(10, 5)

            if event.key == pygame.K_a:
                bal = randint(1, 3)
                if bal == 1:
                    square2.set_text('Русский', 35)
                elif bal == 2:
                    square2.set_text('Яблоко', 35)
                elif bal == 3:
                    square2.set_text('Тетрис', 35)
                square2.draw(10, 5)
    clock.tick(40)
    pygame.display.update()
